% =============================================================================
% \provenance{ids}{proved in}{tested by}{reviewed in}
%
% A small block placed immediately after a statement (after its scope block,
% where present) carrying its campaign bookkeeping. Campaign ids -- claim ids
% from claims/CLAIMS.md, D-numbers from definitions.md -- live ONLY here,
% never inside a definition's or result's descriptive name
% (WRITING-GUIDE.md Rule 1).
% =============================================================================
\newcommand{\provenance}[4]{%
  \par\addvspace{4pt}%
  {\color{AQMprovrule}\noindent\rule{\linewidth}{0.4pt}}\par\vspace{2pt}%
  \noindent\begin{minipage}{\linewidth}
  \small\sffamily
  \textbf{Provenance.}~ids: #1;\quad proved in: #2;\quad tested by: #3;\quad reviewed in: #4.
  \end{minipage}\par\addvspace{6pt}%
}

% =============================================================================
% Status labels. Rendered text is EXACTLY the status word used in
% claims/CLAIMS.md (CLAUDE.md L4/L5): PROVED, SKETCH, CONJECTURE, REFUTED.
% \statusProvedConditional renders the SAME word "PROVED" (the status column
% never grows a fifth word) marked with a dagger; the condition itself
% belongs in the statement text (PRD.md: "A conditional PROVED must display
% its condition in the statement itself"), never abbreviated into the label.
% =============================================================================
\newcommand{\AQMstatusLabel}[2]{\textsf{\bfseries\small\colorbox{#1}{\textcolor{white}{~#2~}}}}
\newcommand{\statusProved}{\AQMstatusLabel{AQMproved}{PROVED}}
\newcommand{\statusProvedConditional}{\AQMstatusLabel{AQMproved}{PROVED}\textsuperscript{$\dagger$}}
\newcommand{\statusSketch}{\AQMstatusLabel{AQMsketch}{SKETCH}}
\newcommand{\statusConjecture}{\AQMstatusLabel{AQMconjecture}{CONJECTURE}}
\newcommand{\statusRefuted}{\AQMstatusLabel{AQMrefuted}{REFUTED}}
